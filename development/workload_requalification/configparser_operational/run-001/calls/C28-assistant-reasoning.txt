class MultilineContinuationErrorTests(unittest.TestCase):
    """Tests for MultilineContinuationError when allow_no_value=True."""

    INPUT = "[service]\noptional_flag\n    continuation\n"

    def _make_parser(self, **kwargs):
        kwargs.setdefault('allow_no_value', True)
        return configparser.ConfigParser(**kwargs)

    def _make_raw_parser(self, **kwargs):
        kwargs.setdefault('allow_no_value', True)
        return configparser.RawConfigParser(**kwargs)

    def test_error_raised_read_string(self):
        parser = self._make_parser()
        with self.assertRaises(configparser.MultilineContinuationError) as cm:
            parser.read_string(self.INPUT)
        self.assertIn('service', str(cm.exception))
        self.assertEqual(cm.exception.line, '    continuation')
        self.assertEqual(cm.exception.lineno, 3)

    def test_error_raised_read_string_raw(self):
        parser = self._make_raw_parser()
        with self.assertRaises(configparser.MultilineContinuationError):
            parser.read_string(self.INPUT)

    def test_error_attributes(self):
        parser = self._make_parser()
        with self.assertRaises(configparser.MultilineContinuationError) as cm:
            parser.read_string(self.INPUT)
        exc = cm.exception
        self.assertEqual(exc.lineno, 3)
        self.assertEqual(exc.line, '    continuation')
        self.assertIsInstance(exc, configparser.ParsingError)

    def test_error_is_parsing_error(self):
        self.assertTrue(issubclass(configparser.MultilineContinuationError,
                                    configparser.ParsingError))

    def test_valid_multiline_still_works(self):
        parser = self._make_parser()
        text = "[sec]\nkey = first line\n    second line\n"
        parser.read_string(text)
        self.assertEqual(parser.get('sec', 'key'), 'first line second line')

    def test_valueless_option_without_continuation(self):
        parser = self._make_parser()
        text = "[sec]\nflag\n"
        parser.read_string(text)
        self.assertIsNone(parser.get('sec', 'flag', raw=True))
