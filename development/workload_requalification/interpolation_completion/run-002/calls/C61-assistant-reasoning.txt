def test_interpolation_missing_option_error_transport(self):
    """Test that InterpolationMissingOptionError survives copy, deepcopy, pickle."""
    cf = self.get_interpolation_config()
    try:
        cf.get("Interpolation Error", "name")
    except configparser.InterpolationMissingOptionError as e:
        exc = e
    # Check basic attributes
    self.assertIsInstance(exc, configparser.InterpolationMissingOptionError)
    self.assertEqual(exc.option, "name")
    self.assertEqual(exc.section, "Interpolation Error")
    self.assertEqual(exc.reference, "reference")
    
    # Shallow copy
    import copy
    exc_copy = copy.copy(exc)
    self.assertEqual(exc_copy.option, "name")
    self.assertEqual(exc_copy.section, "Interpolation Error")
    self.assertEqual(exc_copy.reference, "reference")
    
    # Deep copy
    exc_deep = copy.deepcopy(exc)
    self.assertEqual(exc_deep.option, "name")
    self.assertEqual(exc_deep.section, "Interpolation Error")
    self.assertEqual(exc_deep.reference, "reference")
    
    # Pickle

    for protocol in range(pickle.HIGHEST_PROTOCOL + 1):
