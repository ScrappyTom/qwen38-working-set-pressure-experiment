    def test_urlsplit_port_validation(self):
        # Construction (urlsplit) always succeeds; only .port access raises.
        # --- text URLs ---
        # no port
        p = urllib.parse.urlsplit("http://example.com/")
        self.assertIsNone(p.port)

        # empty port
        p = urllib.parse.urlsplit("http://example.com:/")
        self.assertIsNone(p.port)

        # port 0
        p = urllib.parse.urlsplit("http://example.com:0/")
        self.assertEqual(p.port, 0)

        # port 65535
        p = urllib.parse.urlsplit("http://example.com:65535/")
        self.assertEqual(p.port, 65535)

        # port 65536: out of range
        p = urllib.parse.urlsplit("http://example.com:65536/")
        with self.assertRaises(ValueError) as ctx:
            p.port
        self.assertIs(type(ctx.exception), ValueError)
