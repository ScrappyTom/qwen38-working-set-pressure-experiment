    def test_port_access_boundary_and_errors(self):
        # The port property validates on access, not at construction time.
        for func in (urllib.parse.urlsplit, urllib.parse.urlparse):
            # No port
            p = func("http://example.com/")
            self.assertIs(p.port, None)
            p = func(b"http://example.com/")
            self.assertIs(p.port, None)

            # Empty port (trailing colon, no digits)
            p = func("http://example.com:")
            self.assertIs(p.port, None)
            p = func(b"http://example.com:")
            self.assertIs(p.port, None)

            # Port 0
            p = func("http://example.com:0/")
            self.assertEqual(p.port, 0)
            p = func(b"http://example.com:0/")
            self.assertEqual(p.port, 0)

            # Port 65535 (max valid)
            p = func("http://example.com:65535/")
            self.assertEqual(p.port, 65535)
            p = func(b"http://example.com:65535/")
            self.assertEqual(p.port, 65535)

            # Port 65536 (out of range); construction succeeds
            p = func("http://example.com:65536/")
            with self.assertRaises(ValueError) as cm:
                p.port
            self.assertIs(type(cm.exception), ValueError)
            self.assertEqual(cm.exception.args,
                             ("Port out of range 0-65535",))
            self.assertEqual(str(cm.exception),
                             "Port out of range 0-65535")
            p = func(b"http://example.com:65536/")
            with self.assertRaises(ValueError) as cm:
                p.port
            self.assertIs(type(cm.exception), ValueError)
            self.assertEqual(cm.exception.args,
                             ("Port out of range 0-65535",))

            # Negative port
            p = func("http://example.com:-1/")
            with self.assertRaises(ValueError) as cm:
                p.port
            self.assertIs(type(cm.exception), ValueError)
            self.assertEqual(cm.exception.args,
                             ("Port could not be cast to integer value as '-1'",))
            p = func(b"http://example.com:-1/")
            with self.assertRaises(ValueError) as cm:
                p.port
            self.assertIs(type(cm.exception), ValueError)
            self.assertEqual(cm.exception.args,
                             ("Port could not be cast to integer value as b'-1'",))

            # Non-integer port
            p = func("http://example.com:abc/")
            with self.assertRaises(ValueError) as cm:
                p.port
            self.assertIs(type(cm.exception), ValueError)
            self.assertEqual(cm.exception.args,
                             ("Port could not be cast to integer value as 'abc'",))
            p = func(b"http://example.com:abc/")
            with self.assertRaises(ValueError) as cm:
                p.port
            self.assertIs(type(cm.exception), ValueError)
            self.assertEqual(cm.exception.args,
                             ("Port could not be cast to integer value as b'abc'",))

            # Non-ASCII decimal digit (text URL)
            p = func("http://example.com:\u0661/")
            with self.assertRaises(ValueError) as cm:
                p.port
            self.assertIs(type(cm.exception), ValueError)
            self.assertEqual(cm.exception.args,
                             ("Port could not be cast to integer value as '\u0661'",))
