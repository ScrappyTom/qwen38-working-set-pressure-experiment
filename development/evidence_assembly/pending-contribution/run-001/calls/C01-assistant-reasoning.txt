class InterpolationMissingOptionErrorCopyTestCase(unittest.TestCase):
    """InterpolationMissingOptionError obtained via real get() calls:
    copy, deepcopy, pickle, attributes, raw retrieval, and resolution."""

    def _basic(self):
        parser = configparser.ConfigParser()
        parser.read_string(
            '[sec]\n'
            'val = %(missing)s\n'
        )
        try:
            parser.get('sec', 'val')
        except configparser.InterpolationMissingOptionError as e:
            return parser, e
        self.fail('expected InterpolationMissingOptionError')

    def _extended_same(self):
        parser = configparser.ConfigParser(
            interpolation=configparser.ExtendedInterpolation())
        parser.read_string(
            '[sec]\n'
            'val = ${missing}\n'
        )
        try:
            parser.get('sec', 'val')
        except configparser.InterpolationMissingOptionError as e:
            return parser, e
        self.fail('expected InterpolationMissingOptionError')

    def _extended_cross(self):
        parser = configparser.ConfigParser(
            interpolation=configparser.ExtendedInterpolation())
        parser.read_string(
            '[other]\n'
            '[sec]\n'
            'val = ${other:missing}\n'
        )
        try:
            parser.get('sec', 'val')
        except configparser.InterpolationMissingOptionError as e:
            return parser, e
        self.fail('expected InterpolationMissingOptionError')

    def _check_attrs(self, e, option, section, rawval, reference):
        self.assertIs(type(e), configparser.InterpolationMissingOptionError)
        self.assertEqual(e.option, option)
        self.assertEqual(e.section, section)
        self.assertEqual(e.reference, reference)
        self.assertEqual(e.args, (option, section, rawval, reference))
        expected_msg = (
            'Bad value substitution: option {!r} in section {!r} '
            'contains an interpolation key {!r} which is not a valid '
            'option name. Raw value: {!r}'
            .format(option, section, reference, rawval))
        self.assertEqual(e.message, expected_msg)

    def _check_copy(self, e1, proto='copy'):
        import copy
        if proto == 'copy':
            e2 = copy.copy(e1)
        elif proto == 'deepcopy':
            e2 = copy.deepcopy(e1)
        else:
            import pickle
            e2 = pickle.loads(pickle.dumps(e1, int(proto)))
        self.assertIs(type(e2), configparser.InterpolationMissingOptionError)
        self.assertEqual(e2.option, e1.option)
        self.assertEqual(e2.section, e1.section)
        self.assertEqual(e2.reference, e1.reference)
        self.assertEqual(e2.args, e1.args)
        self.assertEqual(e2.message, e1.message)

        parser, e = self._basic()
        self._check_attrs(e, 'val', 'sec', '%(missing)s', 'missing')
        # raw retrieval
        self.assertEqual(parser.get('sec', 'val', raw=True), '%(missing)s')
        # successful lookup after adding the setting
        parser.set('sec', 'missing', 'resolved')
        self.assertEqual(parser.get('sec', 'val'), 'resolved')
